ЗАДАЧА: Реализовать Secure Deal Room для закрытых сделок с ролью investor и защитой через Supabase

У МЕНЯ УЖЕ ЕСТЬ:
- Supabase Auth (login, роли, approval)
- таблица yachts (с is_private)
- пользователи с ролями (investor, broker, owner, admin)

-----------------------------------
1. СОЗДАТЬ ТАБЛИЦУ DEALS
-----------------------------------

create table if not exists deals (
  id uuid default uuid_generate_v4() primary key,
  yacht_id uuid references yachts(id),
  title text,
  description text,
  market_price text,
  deal_price text,
  location text,
  status text default 'active',
  created_at timestamp default now()
);

-----------------------------------
2. СОЗДАТЬ ТАБЛИЦУ ДОКУМЕНТОВ
-----------------------------------

create table if not exists deal_documents (
  id uuid default uuid_generate_v4() primary key,
  deal_id uuid references deals(id),
  name text,
  file_url text,
  created_at timestamp default now()
);
